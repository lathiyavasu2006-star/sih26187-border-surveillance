# Week 3 ML pipeline dependencies (install after requirements.txt).
#
# PyTorch must come from the CUDA 12.1 index, never plain PyPI (that build is CPU-only):
#   pip install torch==2.5.1+cu121 torchvision==0.20.1+cu121 torchaudio==2.5.1+cu121 --index-url https://download.pytorch.org/whl/cu121
#
# Install ultralytics AFTER torch, otherwise pip pulls the CPU torch wheel from PyPI.
ultralytics==8.3.0
# lap 0.4.0 has no wheel for Python 3.11 on Windows and its setup.py no longer builds;
# 0.5.12 is the same package (import name `lap`) with prebuilt wheels.
lap==0.5.12
websockets==12.0
httpx==0.28.1
onnx==1.17.0

#   tensorrt_cu12_bindings-10.0.1-cp311-none-win_amd64.whl    1e4f708cbd864668385493dee8958ba4e538b2c54172239fd2f64ca45d6ca303
#   tensorrt-cu12-10.0.1.tar.gz (import shim)                 ebb89f6f9b6d187265f3bc829e38cff6e27059508e9a160e229506e5d9b648a0
